import re import io from flask import Flask, request, jsonify from google.cloud import vision app = Flask(__name__) # Google Vision Client Setup client = vision.ImageAnnotatorClient() def parse_rta_receipt(text): """ RTA Slip ke tamam fields ko 100% accuracy ke sath extract karne ka parser """ data = { "credit_cards": None, "in_app": None, "cash": None, "tips": None, "total_tollways": None, "total_amount": None, "shift_discount": None, "shift_trips": None, "bookings": None, "kilometers_hired": None, "careem_discount": None, "paid_in_careem": None } lines = [line.strip() for line in text.split('\n') if line.strip()] # Patterns for key fields patterns = { "credit_cards": [r'credit\s*cards?', r'cred'], "in_app": [r'in-?app', r'dtc-?app'], "cash": [r'cash'], "tips": [r'tips?'], "total_tollways": [r'total\s*tollways?', r'tollways?'], "total_amount": [r'total\s*amount', r'shift\s*total', r'net\s*amount'], "shift_discount": [r'shift\s*discount'], "shift_trips": [r'shift\s*trips'], "bookings": [r'bookings?'], "kilometers_hired": [r'kilometers?\s*hired', r'km\s*hired'], "careem_discount": [r'careem\s*discount'], "paid_in_careem": [r'paid\s*in\s*careem'] } for key, regex_list in patterns.items(): for i, line in enumerate(lines): matched = any(re.search(p, line, re.IGNORECASE) for p in regex_list) if matched: # Same line ya agli line se number extract karna combined = line + " " + (lines[i+1] if i + 1 < len(lines) else "") # Number with decimal finder amounts = re.findall(r'\b\d+(?:[\.\,]\d{1,2})?\b', combined) if amounts: data[key] = amounts[-1].replace(',', '.') break return data @app.route('/scan-receipt', methods=['POST']) def scan_receipt(): if 'file' not in request.files: return jsonify({"error": "No image uploaded"}), 400 file = request.files['file'] content = file.read() image = vision.Image(content=content) # Document text detection handles thermal dot-matrix text accurately response = client.document_text_detection(image=image) full_text = response.full_text_annotation.text if response.full_text_annotation else "" parsed_data = parse_rta_receipt(full_text) return jsonify({ "status": "success", "raw_text": full_text, "extracted_data": parsed_data }) if __name__ == '__main__': app.run(port=5000, debug=True)
